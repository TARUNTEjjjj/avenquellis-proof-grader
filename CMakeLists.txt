cmake_minimum_required(VERSION 3.20)
project(VeyraGrader VERSION 0.8.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
option(NOETHER_GRADER_BUILD_CLI "Build the native JSON and HTTP client" ON)
option(NOETHER_GRADER_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
option(NOETHER_GRADER_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(CMAKE_CXX_EXTENSIONS OFF)

function(noether_configure target)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        if(NOETHER_GRADER_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
        if(NOETHER_GRADER_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PUBLIC -fsanitize=address,undefined)
        endif()
    endif()
endfunction()

add_library(noether_grader STATIC src/grader.cpp src/certificate.cpp)
add_library(NoetherGrader::grader ALIAS noether_grader)
add_library(VeyraGrader::grader ALIAS noether_grader)
set_target_properties(noether_grader PROPERTIES EXPORT_NAME grader)
target_compile_features(noether_grader PUBLIC cxx_std_20)
target_include_directories(noether_grader PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
noether_configure(noether_grader)

if(NOETHER_GRADER_BUILD_CLI)
    find_package(CURL 7.62 REQUIRED)
    # A static libcurl returned by FindCURL may require the additional libraries in its pkg-config record.
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(NOETHER_CURL QUIET IMPORTED_TARGET libcurl>=7.62)
    endif()
    add_executable(veyra-grader app/main.cpp app/transport.cpp src/json_api.cpp)
    target_include_directories(veyra-grader PRIVATE src)
    target_include_directories(veyra-grader SYSTEM PRIVATE third_party)
    if(TARGET PkgConfig::NOETHER_CURL)
        target_link_libraries(veyra-grader PRIVATE NoetherGrader::grader PkgConfig::NOETHER_CURL)
    else()
        target_link_libraries(veyra-grader PRIVATE NoetherGrader::grader CURL::libcurl)
    endif()
    noether_configure(veyra-grader)
    add_custom_command(TARGET veyra-grader POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_FILE:veyra-grader> $<TARGET_FILE_DIR:veyra-grader>/noether-grader${CMAKE_EXECUTABLE_SUFFIX})
    add_custom_target(noether-grader DEPENDS veyra-grader)
    install(TARGETS veyra-grader RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(PROGRAMS $<TARGET_FILE:veyra-grader> DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME noether-grader${CMAKE_EXECUTABLE_SUFFIX})
endif()

if(BUILD_TESTING)
    add_executable(grader-native-tests native-tests/core.cpp)
    target_link_libraries(grader-native-tests PRIVATE NoetherGrader::grader)
    noether_configure(grader-native-tests)
    add_test(NAME native-core COMMAND grader-native-tests)
    if(NOT IS_ABSOLUTE "${CMAKE_INSTALL_INCLUDEDIR}" AND NOT IS_ABSOLUTE "${CMAKE_INSTALL_LIBDIR}" AND NOT IS_ABSOLUTE "${CMAKE_INSTALL_BINDIR}")
        add_test(NAME installed-consumer COMMAND ${CMAKE_COMMAND} -DBUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}
            -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR} -DCOMPILER=${CMAKE_CXX_COMPILER} -DCONFIG=$<CONFIG>
            -DCONFIG_UPPER=$<UPPER_CASE:$<CONFIG>> -DEXE_SUFFIX=${CMAKE_EXECUTABLE_SUFFIX} -P ${CMAKE_CURRENT_SOURCE_DIR}/native-tests/install.cmake)
    endif()
    find_program(NOETHER_NODE node)
    if(NOETHER_GRADER_BUILD_CLI AND NOETHER_NODE)
        add_test(NAME certificate-parity COMMAND ${NOETHER_NODE} ${CMAKE_CURRENT_SOURCE_DIR}/native-tests/parity.mjs $<TARGET_FILE:veyra-grader>)
        add_test(NAME review-transport COMMAND ${NOETHER_NODE} ${CMAKE_CURRENT_SOURCE_DIR}/native-tests/transport.mjs $<TARGET_FILE:veyra-grader>)
        set_tests_properties(certificate-parity review-transport PROPERTIES TIMEOUT 120)
    endif()
endif()

install(TARGETS noether_grader EXPORT NoetherGraderTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
set(package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/NoetherGrader)
install(EXPORT NoetherGraderTargets NAMESPACE NoetherGrader:: DESTINATION ${package_dir})
configure_package_config_file(cmake/NoetherGraderConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/NoetherGraderConfig.cmake INSTALL_DESTINATION ${package_dir})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/NoetherGraderConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/NoetherGraderConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/NoetherGraderConfigVersion.cmake DESTINATION ${package_dir})
if(NOETHER_GRADER_BUILD_CLI)
    install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/NoetherGrader RENAME nlohmann-json-LICENSE.MIT)
endif()

# The canonical Veyra package and its original installed name resolve the same ABI.
set(veyra_package_dir ${CMAKE_INSTALL_LIBDIR}/cmake/VeyraGrader)
configure_package_config_file(cmake/VeyraGraderConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/VeyraGraderConfig.cmake
    INSTALL_DESTINATION ${veyra_package_dir})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/VeyraGraderConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/VeyraGraderConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/VeyraGraderConfigVersion.cmake DESTINATION ${veyra_package_dir})
